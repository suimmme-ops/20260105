    <script>
        // --- State Management ---
        let rootData = null;
        let svg, g, tree, root;
        let i = 0;
        const duration = 500;
        
        // --- API Logic ---
        
        // Helper function for raw fetch to handle retries cleanly
        async function executeGeminiCall(modelName, key, promptText) {
             const url = `https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${key}`;
             const response = await fetch(url, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    contents: [{ parts: [{ text: promptText }] }]
                })
            });
            return response;
        }

        async function fetchAnalysis(text) {
            const activeKey = apiKey || fallbackKey;
            
            // Priority Updated:
            // Always try gemini-2.5-flash-preview-09-2025 first.
            // If that fails (404/503), fall back to gemini-pro.
            
            let primaryModel = "gemini-2.5-flash-preview-09-2025";
            const backupModel = "gemini-pro";

            if (!activeKey) {
                alert("API 키가 설정되지 않았습니다. 코드 상단의 fallbackKey 변수에 API 키를 입력해주세요.");
                return null;
            }

            const prompt = `
                Analyze the following "What if" scenario or statement: "${text}".
                Identify exactly 3 distinct PROS (positive outcomes/advantages) and 3 distinct CONS (negative outcomes/disadvantages).
                
                IMPORTANT: Return ONLY a valid JSON object. Do not include Markdown formatting, code blocks, or extra text.
                Format:
                {
                    "pros": ["Pro 1", "Pro 2", "Pro 3"],
                    "cons": ["Con 1", "Con 2", "Con 3"]
                }
                Keep the sentences concise (under 10 words if possible) and in Korean.
            `;

            try {
                let response = await executeGeminiCall(primaryModel, activeKey, prompt);

                // If primary fails with 404, try backup
                if (response.status === 404) {
                    console.warn(`Model ${primaryModel} not found. Retrying with ${backupModel}...`);
                    response = await executeGeminiCall(backupModel, activeKey, prompt);
                }

                if (!response.ok) {
                    const errText = await response.text();
                    console.error("API Error Detail:", errText);
                    throw new Error(`API Error: ${response.status} (${response.statusText})`);
                }

                const data = await response.json();
                
                if (!data.candidates || data.candidates.length === 0) {
                     throw new Error("API 응답에 내용이 없습니다. (Safety filter triggered?)");
                }

                const rawText = data.candidates[0].content.parts[0].text;
                const jsonText = rawText.replace(/```json|```/g, '').trim();
                return JSON.parse(jsonText);

            } catch (error) {
                console.error("Gemini API Error:", error);
                alert(`분석 중 오류가 발생했습니다.\n\n상세 내용: ${error.message}\n\nAPI 키와 네트워크 상태를 확인해주세요.`);
                return null;
            }
        }

        // --- Core UI Logic ---

        function startMindMap() {
            const input = document.getElementById('initial-input');
            const query = input.value.trim();
            if (!query) return;

            // UI Transition
            const landing = document.getElementById('landing-page');
            const mapPage = document.getElementById('mindmap-page');
            const headerTitle = document.getElementById('header-title');
            const loader = document.getElementById('global-loader');

            landing.classList.add('fade-exit-active');
            
            setTimeout(async () => {
                landing.classList.add('hidden');
                mapPage.classList.remove('hidden');
                // Trigger reflow
                void mapPage.offsetWidth; 
                mapPage.classList.remove('opacity-0');
                
                headerTitle.textContent = query;
                loader.classList.remove('hidden'); // Show loader

                // Initial API Call
                const result = await fetchAnalysis(query);
                
                if (result) {
                    initD3(query, result);
                } else {
                    resetApp(); // Fail gracefully
                }
                
                loader.classList.add('hidden');
            }, 500);
        }

        function resetApp() {
            location.reload();
        }

        // --- D3.js Visualization Logic ---

        function initD3(query, initialData) {
            const container = document.getElementById('canvas-container');
            const width = container.clientWidth;
            const height = container.clientHeight;

            // 1. Data Structure Construction
            rootData = {
                name: query,
                type: 'root',
                children: [
                    ...initialData.pros.map(t => ({ name: t, type: 'pro', children: [] })),
                    ...initialData.cons.map(t => ({ name: t, type: 'con', children: [] }))
                ]
            };

            // 2. SVG Setup
            d3.select("#canvas-container").selectAll("*").remove(); // Clear previous

            svg = d3.select("#canvas-container").append("svg")
                .attr("width", width)
                .attr("height", height)
                .call(d3.zoom().on("zoom", (e) => {
                    g.attr("transform", e.transform);
                }))
                .append("g");
            
            g = svg.append("g").attr("transform", "translate(" + width / 4 + "," + height / 2 + ")");

            tree = d3.tree().nodeSize([40, 250]); // [height-spacing, width-spacing]

            root = d3.hierarchy(rootData, d => d.children);
            root.x0 = 0;
            root.y0 = 0;

            // Initial Update
            update(root);
        }

        async function update(source) {
            const treeData = tree(root);

            // Compute the new tree layout.
            const nodes = treeData.descendants();
            const links = treeData.links();

            // Normalize for fixed-depth.
            nodes.forEach(d => { d.y = d.depth * 280; }); // Width between levels

            // ****************** Nodes section ***************************

            const node = g.selectAll('g.node')
                .data(nodes, d => d.id || (d.id = ++i));

            // Enter any new modes at the parent's previous position.
            const nodeEnter = node.enter().append('g')
                .attr('class', 'node')
                .attr("transform", d => "translate(" + source.y0 + "," + source.x0 + ")")
                .on('click', click);

            // Add Circle for the nodes
            nodeEnter.append('circle')
                .attr('class', 'node')
                .attr('r', 1e-6)
                .style("fill", d => {
                    if (d.data.type === 'pro') return "#E3F2FD"; // Light Blue
                    if (d.data.type === 'con') return "#FFEBEE"; // Light Red
                    return "#374151"; // Root Gray
                })
                .style("stroke", d => {
                    if (d.data.type === 'pro') return "#2196F3"; // Blue
                    if (d.data.type === 'con') return "#F44336"; // Red
                    return "#111827"; // Root Dark
                });

            // Add Text
            nodeEnter.append('text')
                .attr("dy", ".35em")
                .attr("x", d => d.children || d._children ? -15 : 15)
                .attr("text-anchor", d => d.children || d._children ? "end" : "start")
                .text(d => d.data.name)
                .style("fill-opacity", 1e-6)
                .style("fill", "#374151") // Gray-700
                .call(wrap, 200); // Wrap text function

            // Loading Spinner (hidden by default)
            const spinner = nodeEnter.append("g")
                .attr("class", "loading-spinner")
                .attr("display", "none");
            
            spinner.append("path")
                .attr("d", "M10,0 A10,10 0 0,1 0,10") // Quarter circle
                .style("fill", "none")
                .style("stroke", "#2563EB")
                .style("stroke-width", "2px");

            // UPDATE
            const nodeUpdate = node.merge(nodeEnter).transition()
                .duration(duration)
                .attr("transform", d => "translate(" + d.y + "," + d.x + ")");

            nodeUpdate.select('circle')
                .attr('r', 8)
                .style("fill", d => d._children ? "#fff" : (
                    d.data.type === 'pro' ? "#E3F2FD" : (d.data.type === 'con' ? "#FFEBEE" : "#374151")
                ))
                .attr('cursor', 'pointer');

            nodeUpdate.select('text')
                .style("fill-opacity", 1);

            // EXIT
            const nodeExit = node.exit().transition()
                .duration(duration)
                .attr("transform", d => "translate(" + source.y + "," + source.x + ")")
                .remove();

            nodeExit.select('circle')
                .attr('r', 1e-6);

            nodeExit.select('text')
                .style("fill-opacity", 1e-6);

            // ****************** Links section ***************************

            const link = g.selectAll('path.link')
                .data(links, d => d.target.id);

            // Enter any new links at the parent's previous position.
            const linkEnter = link.enter().insert('path', "g")
                .attr("class", "link")
                .attr('d', d => {
                    const o = {x: source.x0, y: source.y0};
                    return diagonal(o, o);
                })
                .style("stroke", d => {
                    // Gradient-like coloring for links based on target type
                    if (d.target.data.type === 'pro') return "#90CAF9";
                    if (d.target.data.type === 'con') return "#EF9A9A";
                    return "#ccc";
                });

            // UPDATE
            const linkUpdate = link.merge(linkEnter).transition()
                .duration(duration)
                .attr('d', d => diagonal(d.source, d.target));

            // EXIT
            link.exit().transition()
                .duration(duration)
                .attr('d', d => {
                    const o = {x: source.x, y: source.y};
                    return diagonal(o, o);
                })
                .remove();

            // Store the old positions for transition.
            nodes.forEach(d => {
                d.x0 = d.x;
                d.y0 = d.y;
            });
        }

        // --- Helper: Diagonal Generator for Curves ---
        function diagonal(s, d) {
            return `M ${s.y} ${s.x}
                    C ${(s.y + d.y) / 2} ${s.x},
                      ${(s.y + d.y) / 2} ${d.x},
                      ${d.y} ${d.x}`;
        }

        // --- Helper: Text Wrap ---
        function wrap(text, width) {
            text.each(function() {
                var text = d3.select(this),
                    words = text.text().split(/\s+/).reverse(),
                    word,
                    line = [],
                    lineNumber = 0,
                    lineHeight = 1.2, // ems
                    y = text.attr("y"),
                    dy = parseFloat(text.attr("dy")),
                    tspan = text.text(null).append("tspan").attr("x", d => d.children || d._children ? -15 : 15).attr("y", y).attr("dy", dy + "em");
                
                // For simplified wrapping in node logic (very basic)
                // In a real app, complex splitting is needed. 
                // Here we just limit char length per line for simplicity
                const fullText = words.reverse().join(" ");
                if (fullText.length > 20) {
                   // Split roughly
                   const mid = Math.floor(fullText.length / 2);
                   const part1 = fullText.substring(0, mid);
                   const part2 = fullText.substring(mid);
                   tspan.text(part1 + "-");
                   text.append("tspan").attr("x", d => d.children || d._children ? -15 : 15).attr("y", y).attr("dy", ++lineNumber * lineHeight + dy + "em").text(part2);
                } else {
                    tspan.text(fullText);
                }
            });
        }

        // --- Interaction Logic: Click to Expand ---
        async function click(event, d) {
            if (d.children) {
                // Already expanded -> Collapse
                d._children = d.children;
                d.children = null;
                update(d);
            } else if (d._children) {
                 // Collapsed -> Expand (from cache)
                d.children = d._children;
                d._children = null;
                update(d);
            } else {
                // No children -> Fetch Data!
                
                // 1. Show Spinner
                const gNode = d3.select(this);
                const spinner = gNode.select(".loading-spinner");
                spinner.style("display", "block").classed("spinner", true);
                
                // 2. API Call
                const result = await fetchAnalysis(d.data.name);
                
                spinner.style("display", "none").classed("spinner", false);

                if (result) {
                    const newChildren = [
                        ...result.pros.map(t => ({ name: t, type: 'pro', children: [] })),
                        ...result.cons.map(t => ({ name: t, type: 'con', children: [] }))
                    ];
                    
                    // Add to data model
                    if (!d.data.children) d.data.children = [];
                    d.data.children = newChildren; // update raw data
                    
                    // Add to hierarchy
                    const newNodeNodes = newChildren.map(childData => {
                        const n = d3.hierarchy(childData);
                        n.depth = d.depth + 1;
                        n.height = d.height - 1;
                        n.parent = d;
                        return n;
                    });
                    
                    d.children = newNodeNodes;
                    
                    // Expand and Update
                    update(d);
                }
            }
        }

        // Add Enter Key Support
        document.getElementById("initial-input").addEventListener("keypress", function(event) {
            if (event.key === "Enter") {
                startMindMap();
            }
        });
